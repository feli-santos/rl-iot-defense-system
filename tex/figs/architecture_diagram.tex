\documentclass[border=4pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, fit, backgrounds, calc}

% ---- colour palette (greyscale-friendly) --------------------
\definecolor{colOffline}{RGB}{220,235,245}   % light blue  – offline stages
\definecolor{colOnline}{RGB}{220,245,220}    % light green – online RL loop
\definecolor{colBench}{RGB}{245,235,215}     % light amber – benchmark
\definecolor{colBorder}{RGB}{80,100,120}     % dark slate  – borders
\definecolor{colArrow}{RGB}{60,80,100}

% ---- common node styles -------------------------------------
\tikzset{
  box/.style={
    rectangle, rounded corners=3pt,
    draw=colBorder, line width=0.7pt,
    text centered, font=\small,
    inner sep=5pt, minimum height=1.8em
  },
  stageBox/.style={
    box,
    minimum width=3.6cm, minimum height=2.6cm,
    align=center
  },
  envBox/.style={
    box,
    minimum width=3.2cm, minimum height=2.0cm,
    align=center
  },
  groupLabel/.style={
    font=\small\bfseries, text=colBorder
  },
  arr/.style={
    -{Stealth[length=5pt]}, line width=0.8pt, color=colArrow
  },
  dasharr/.style={
    arr, dashed
  }
}

\begin{document}
\begin{tikzpicture}[node distance=0.45cm and 0.55cm]

%% ==============================================================
%%  STAGE I  — three offline preparation lanes
%% ==============================================================

% --- Lane 1: Dataset pipeline ---
\node[stageBox, fill=colOffline] (dataset) {
  \textbf{Dataset Pipeline}\\[3pt]
  CICIoT2023\\
  (105 IoT devices,\\
   33 attack types)\\[3pt]
  Kill-chain projection\\
  ($\times$5 stages)\\[3pt]
  Splits: train / val /\\
  test / ood
};

% --- Lane 2: Red-Team LSTM ---
\node[stageBox, fill=colOffline, right=of dataset] (redteam) {
  \textbf{Red-Team LSTM}\\[3pt]
  Next-stage token model\\
  over kill-chain\\
  sequences\\[3pt]
  Upper-triangular\\
  monotonic mask\\[3pt]
  Trained on\\
  stage tokens (train)
};

% --- Lane 3: Stage Detector ---
\node[stageBox, fill=colOffline, right=of redteam] (detector) {
  \textbf{Stage Detector}\\[3pt]
  MLP (production)\\
  RandomForest\\
  1D-CNN\\[3pt]
  Input: 29-dim\\
  per-flow features\\[3pt]
  Output: $\hat{s} \in$\\
  \{benign, \ldots, impact\}
};

% --- Offline bounding box ---
\begin{scope}[on background layer]
  \node[draw=colBorder, fill=colOffline!40, rounded corners=5pt,
        inner sep=8pt,
        fit=(dataset)(detector),
        label={[groupLabel]above:Stage I — Offline Preparation (dashed boxes)}] (stageI) {};
\end{scope}

%% ==============================================================
%%  STAGE II  — Online RL loop
%% ==============================================================

% --- Adversarial Environment ---
\node[envBox, fill=colOnline,
      below=1.1cm of redteam] (env) {
  \textbf{AdversarialIoTEnv}\\[3pt]
  Hidden attack stage $s_t$\\
  Kill-chain MDP\\
  $(S,A,P,R,\gamma)$
};

% --- Realization Engine ---
\node[envBox, fill=colOnline,
      left=0.7cm of env] (realizer) {
  \textbf{Realization Engine}\\[3pt]
  Split-aware sampling\\
  of 29-dim feature\\
  vectors from train
};

% --- Reward function ---
\node[envBox, fill=colOnline,
      right=0.7cm of env] (reward) {
  \textbf{Reward Function}\\[3pt]
  Stage-action\\
  proportionality\\
  + de-escalation\\
  + terminal-impact
};

% --- RL Agent ---
\node[envBox, fill=colOnline,
      below=0.8cm of env] (agent) {
  \textbf{Blue-Team Agent}\\[3pt]
  DQN / PPO / A2C\\
  (Stable Baselines3\\
   MultiInputPolicy)\\[3pt]
  10 seeds $\times$ 250k steps
};

% --- Online bounding box ---
\begin{scope}[on background layer]
  \node[draw=colBorder, fill=colOnline!40, rounded corners=5pt,
        inner sep=10pt,
        fit=(realizer)(reward)(agent),
        label={[groupLabel]above:Stage II — Online RL Training Loop (solid box)}] (stageII) {};
\end{scope}

%% ==============================================================
%%  STAGE III  — Benchmark & Ablation
%% ==============================================================

\node[box, fill=colBench,
      minimum width=11.6cm, minimum height=2.2cm,
      align=center,
      below=1.0cm of stageII.south] (bench) {
  \textbf{Stage III — Benchmark \& Ablation (held-out \texttt{test\_balanced})}\\[4pt]
  Trained DRL (DQN / PPO / A2C) \quad vs.\quad
  Random \;|\; Always-Observe \;|\; Always-Block \;|\; RF-Acting \;|\; Oracle\\[2pt]
  Reward-component ablation (F9) \;|\;
  Aggressiveness sweep (F10) \;|\;
  OOD injection evaluation (F15)
};

%% ==============================================================
%%  ARROWS
%% ==============================================================

% Stage I → Stage II (dataset to realizer)
\draw[arr] (dataset.south) -- ++(0,-0.45) -| (realizer.north);

% Stage I → Stage II (LSTM to env)
\draw[arr] (redteam.south) -- (env.north);

% Stage I → Stage II (detector to reward, dashed = eval-time only)
\draw[dasharr] (detector.south) -- ++(0,-0.45) -| (reward.north)
    node[midway, right, font=\tiny, text=colBorder] {eval-time};

% Realizer ↔ Env
\draw[arr] (realizer.east) -- node[above, font=\tiny]{obs} (env.west);

% Env ↔ Reward
\draw[arr] (env.east) -- node[above, font=\tiny]{$(s_t,a_t)$} (reward.west);
\draw[arr] (reward.south) -- ++(0,-0.3) -| (agent.east)
    node[near start, right, font=\tiny]{$r_t$};

% Env → Agent
\draw[arr] (env.south) -- node[right, font=\tiny]{obs} (agent.north);

% Agent → Env (action)
\draw[arr] (agent.west) -- ++(-0.6,0) |-
    node[near end, left, font=\tiny]{$a_t$} (realizer.south);

% Stage II → Stage III
\draw[arr] ([yshift=-2pt]stageII.south) -- (bench.north);

%% ==============================================================
%%  OBSERVATION DICT label
%% ==============================================================
\node[font=\tiny, align=left, text=colBorder,
      below=2pt of env.south east, anchor=north east, xshift=-2pt] {
  obs = \{features, history, action\_hist, [stage\_pred]\}
};

\end{tikzpicture}
\end{document}
